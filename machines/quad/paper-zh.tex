\section{停机判据（每台独立，转移级完整证明）}

\begin{theorem}[cm16]\label{thm:h16}
cm16 停机 $\iff$ 某时刻状态 $D$ 读格 $w$ 之 $0$ 且 $\tau(w{+}1)=1$。
\end{theorem}
\begin{proof}
停机转移为 $F$ 读 $1$（$F1=\HALT$）。$F$ 的唯一入边为 $D0=\texttt{1RF}$：$D$ 于格 $w$ 读 $0$、写 $1$ 于 $w$、右行至 $w{+}1$ 入 $F$；故 $F$ 的每次到达都紧随一次 $D0$，且 $F$ 所读格恰为 $w{+}1$。于是停机发生 $\iff$ 某次 $D0$ 之后 $\tau(w{+}1)=1$，即所述判据。（$D0$ 的触发条件 $\tau(w)=0$ 已含在``$D$ 读格 $w$ 之 $0$''中。）两方向皆由这一条入边链直接给出。
\end{proof}

\begin{theorem}[cm26]\label{thm:h26}
cm26 停机 $\iff$ 某时刻状态 $B$ 读格 $q$ 之 $1$ 且 $\tau(q{-}1)=0$。
\end{theorem}
\begin{proof}
停机转移为 $E$ 读 $0$。$E$ 的唯一入边为 $B1=\texttt{0LE}$：$B$ 于 $q$ 读 $1$、写 $0$ 于 $q$、\textbf{左}行至 $q{-}1$ 入 $E$；$E$ 所读格为 $q{-}1$。停机 $\iff$ 某次 $B1$ 之后 $\tau(q{-}1)=0$。
\end{proof}

\begin{theorem}[cm51]\label{thm:h51}
cm51 停机 $\iff$ 某时刻状态 $C$ 或 $D$ 读格 $w$ 之 $0$ 且 $\tau(w{+}1)=1$。
\end{theorem}
\begin{proof}
停机转移为 $E$ 读 $1$。$E$ 的入边恰两条：$C0=\texttt{1RE}$ 与 $D0=\texttt{0RE}$，二者皆\textbf{右}行（写后至 $w{+}1$ 入 $E$）。故停机 $\iff$ 某次 $C0$ 或 $D0$ 之后 $\tau(w{+}1)=1$。
\end{proof}

\begin{theorem}[cm52]\label{thm:h52}
cm52 停机 $\iff$ 某时刻状态 $C$ 读格 $c$ 之 $0$ 且 $\tau(c{-}1)=0$。
\end{theorem}
\begin{proof}
停机转移为 $E$ 读 $0$。$E$ 的唯一入边为 $C0=\texttt{0LE}$：$C$ 于 $c$ 读 $0$、左行至 $c{-}1$ 入 $E$，$E$ 读 $\tau(c{-}1)$。
\end{proof}

\begin{remark}
四条判据的证明是纯转移级的（单入边或双入边枚举），不依赖任何动力学；校验器的作用是把判据表述与全部构型实例双向核对（防表述笔误），非证明的必要部分。
\end{remark}

\section{lap 文法与时代代数}

\begin{definition}[标记态与时代]
标记态 = 舞族例程结束时的 $(A,h,\tau)$；相邻标记态之间为一个时代。空带起始段经直接模拟进入首次标记。
\end{definition}

\begin{lemma}[标准 lap 字]\label{lem:lap}
对每台机器、每个参数 $R\ge2$，一个 lap 宏步的（状态,读）事件字为
\[
\underbrace{A0\,B0\,C1\,D1\,A1\,E0\,D0\,F0}_{8\ \text{步}}\;\Big|\;
\underbrace{B1^{R-2}}_{R-2\ \text{步}}\;\underbrace{B0\,C0\,C0\,C1\,D1\,A1}_{6\ \text{步}}\;\Big|\;
\underbrace{E1^{R-1}}_{R-1\ \text{步}}\;\underbrace{E0\,D1\,A0}_{3\ \text{步}},
\]
共 $8+(R{-}2)+6+(R{-}1)+3=2R+14$ 个事件（事件 = 一次（状态,读）；末事件 $A0$ 为与下一 lap 共享的边界事件，lap 本体 $2R{+}13$ 次转移）；净效应 $(x,R,z)\mapsto(x{+}1,\,R{-}2,\,z{+}1)$，工作窗整体右移一格、头相对工作窗位置不变。
\end{lemma}
\begin{proof}[cm16 的完整推导；其余三台同法]
设左缝为格 $g,g{+}1$（$0,0$），右库 $1^R$ 占 $g{+}2..g{+}R{+}1$，右缝 $g{+}R{+}2,g{+}R{+}3$，存款 $1^z$ 自 $g{+}R{+}4$。头 $A$ 在 $g$。

\emph{段一（8 步）}：$A0{=}\texttt{1RB}$ 写 $1@g$、右行；$B0{=}\texttt{0RC}$ 于 $g{+}1$；$C1{=}\texttt{1RD}$ 于 $g{+}2$；$D1{=}\texttt{0LA}$ 写 $0@g{+}3$、左行；$A1{=}\texttt{0LE}$ 写 $0@g{+}2$、左行；$E0{=}\texttt{0RD}$ 于 $g{+}1$；$D0{=}\texttt{1RF}$ 写 $1@g{+}2$、右行；$F0{=}\texttt{1RB}$ 写 $1@g{+}3$、右行。此时 $g{+}2,g{+}3=1,1$，头 $B$ 在 $g{+}4$（库内）。段内第 7--8 步间带面出现瞬态孤岛（$g{+}1,g{+}2,g{+}3=0,1,0$），头距 $\le1$。

\emph{段二（$R{+}4$ 步）}：$B1{=}\texttt{1RB}$ 保写右行扫过库内 $g{+}4..g{+}R{+}1$（$R{-}2$ 格）；$B0$ 于右缝首格 $g{+}R{+}2$（$\texttt{0RC}$）；$C0\,C0$ 两次 $\texttt{1LC}$ 写 $1$ 于 $g{+}R{+}3$ 再 $g{+}R{+}2$（左行）；$C1$ 于库末 $g{+}R{+}1$（$\texttt{1RD}$）；$D1$ 写 $0@g{+}R{+}2$；$A1$ 写 $0@g{+}R{+}1$，左行入 $E$ 于 $g{+}R$。

\emph{段三（$R{+}2$ 步）}：$E1{=}\texttt{1LE}$ 保写左行扫回 $g{+}R..g{+}2$（$R{-}1$ 格）；$E0{=}\texttt{0RD}$ 于 $g{+}1$；$D1{=}\texttt{0LA}$ 写 $0@g{+}2$，左行至 $A$ 于 $g{+}1$——下一 lap 起点。

\emph{净效应核对}：终带为 $g{=}1$（左库 $+1$）；新左缝 $g{+}1,g{+}2$；库残段 $g{+}3..g{+}R$（$R{-}2$ 格）；新右缝 $g{+}R{+}1,g{+}R{+}2$；$g{+}R{+}3{=}1$ 并入存款（$+1$）。即 $(x{+}1,\,R{-}2,\,z{+}1)$，下一 lap 头在新左缝左格——相对位置不变，工作窗右移一格。步数 $8+(R{-}2)+6+(R{-}1)+3=2R{+}14$。

\emph{跨机说明}：字是\textbf{事件级}的（状态,读序列）——各台转移的移动方向可不同（如 $D1$ 在 cm16 为 $\texttt{0LA}$ 左行、在 cm26 为 $\texttt{0RA}$ 右行），但事件序列与净簿记一致；逐台几何由 cell-diff 在深扫锚点上验证（合计约 78 万 lap）〔casecheck-island〕。
\end{proof}

\begin{remark}[时代结构（已验证事实）]
逐 lap 簿记由引理~\ref{lem:lap} 给出（每 lap 右库 $-2$）。库存耗尽触发排空与 churn 重置；重置分支的键集（键 =（位置类， 邻域块数））经深扫采集：cm16/26/51 各 78 键逐键相同、cm52 为 89 键。本文不声明时代层的闭式映射——逐键的输出表以 census/era 数据为准（复跑入口见文末）。
\end{remark}

\section{时代系统：数据与结构}

\begin{definition}[储库构型]
标记态的带面为：左储库 $1$-块、右储库 $1$-块与存款块，库间为 $0^2$ 缝，尾部为冻结骨架（每台由深扫确定，逐位相同）；工作三元组 $(x,R,z)$ = 工作块、右库、存款的长度。
\end{definition}

\begin{remark}[时代链实例（cm16，$10^9$ 步附近）]
实测时代链的工作三元组序列：
\[
(5392,\,22850,\,1818)\ \longrightarrow\ (1787,\,16815,\,1)\ \xrightarrow{\ \text{churn}\ }\ (14188,\,11225,\,946)\ \longrightarrow\ \cdots
\]
右库沿 lap 链按步长 2 递减至耗尽，触发排空与 churn 级联（重播种到新储库布局）；链的截断与重置分支构成 era 键集。
\end{remark}

\begin{remark}[跨机对齐数据]
四台的大时代 $(L_0,R_0)$ 序列实测 562 对齐、常量偏移（$0$ 或 $+1$）、零错位（深扫前缀，$5\times10^9$ 步级）；era 键集 cm16/26/51 各 78 键\textbf{逐键相同}，cm52 为 89 键（判据与骨架差异所致）。此为观测事实（B 级），不作为定理使用。
\end{remark}

\section{四台间已验证的对应事实}

\begin{remark}
以下为分层事实，非同构定理（四台转移表各异，转移级同构不存在）：
\begin{enumerate}
\item lap 文法与簿记在四台上同一（引理~\ref{lem:lap}，每台各自验证）；
\item 大时代 $(L_0,R_0)$ 序列 562 对在常量偏移（$0$ 或 $+1$）下零错位——深扫前缀上的观测〔$5\times10^9$ 步级〕；
\item 末端构型仅冻结骨架逐位相同，工作三元组差 $\pm100$（对早期``逐位相同''测量结论的修正）；
\item 停机判据不同源（定理~\ref{thm:h16}--\ref{thm:h52}：四台的停机态、入边与图案方向互不相同），非停机装配逐台独立进行。
\end{enumerate}
\end{remark}

\section{孤岛不变量与危险图案}

\begin{definition}[孤岛]
孤岛 = 图案 $(0^1,1^1,0^1)$（单格 $1$ 岛）。危险事件 = 该图案出现且\textbf{头远离}（与摧毁波不同位）。
\end{definition}

\begin{theorem}[孤岛不变量]\label{thm:island}
在验证域内（每台 $2\times10^7$ 步，892 个孤岛时刻）：孤岛每次出现均与摧毁波同位（头距 $\le1$）；孤岛的出现为瞬态——每个 lap 内孤岛存在于\textbf{两}个段（修正早期``仅边界 $R{=}1$ 段''的说法），且创造者与摧毁者为同一道经过波，存在窗口短于一个 lap〔casecheck-island，39 断言全过〕。
\end{theorem}

\begin{definition}[词表、audit 与 guard]
\textbf{已验证词表} = 深扫与 census 中实际出现并归档的例程分支集（标准 lap 字及其参数分支、排空分支、churn 重播种分支；完整清单以 era 数据与 \texttt{quad.c} 的分类器为准）。\textbf{audit 模式}：在每个 $A$-态落点处检查五格窗口（头邻域 $\pm2$）是否出现危险图案（头远 $(0^1,1^1,0^1)$），逐步执行至给定深度。\textbf{guard 模式}：每一步直接检查该机判据（定理~\ref{thm:h16}--\ref{thm:h52} 的二格条件）是否成立，逐步执行至给定深度。
\end{definition}

\begin{theorem}[词表内安全]\label{thm:vocab}
在已验证词表覆盖的一切 $A$-态落点处，定理~\ref{thm:h16}--\ref{thm:h52} 的图案前提不出现：冲刺落点的右间隙 $\ge2$（落点几何由字结构给出）；孤岛位被定理~\ref{thm:island} 排除；排空分支的落点窗口经 census 逐格核对。
\end{theorem}

\section{归约与开放问题}

\begin{theorem}[归约（每台）]\label{thm:reduction}
对每台机器 $m\in\{16,26,51,52\}$，以下等价：
\begin{enumerate}
\item $m$ 停机；
\item $m$ 的判据图案（定理~\ref{thm:h16}--\ref{thm:h52} 之二格条件）在某个\textbf{词表外}落点处成立（危险事件）。
\end{enumerate}
机器验证：audit 模式对词表外落点逐步五格检查（$10^9$ 步 $\times$4）危险出现 $=0$；guard 模式（$2\times10^{11}$ 步 $\times$4，逐步检查停机前提）零触发；裸机深扫 cm16 至 $5\times10^{11}$ 步零停机。
\end{theorem}
\begin{proof}
(2)$\Rightarrow$(1)：判据图案成立即触发定理~\ref{thm:h16}--\ref{thm:h52} 的入边链，停机。(1)$\Rightarrow$(2)：设停机，则某判据图案被访问；由定理~\ref{thm:vocab}（词表内安全），该访问必在词表覆盖之外的 $A$-态落点处；词表外落点的枚举完备性由 census/era 键集覆盖承载〔A$^-$〕。
\end{proof}

\begin{conjecture}[五格避让]\label{conj:quad}
四台机器的运行中危险事件永不发生（形式化：定理~\ref{thm:reduction}(2) 恒假）。
\end{conjecture}

\begin{theorem}[局部抽象否证]\label{thm:kclosure}
对 cm16 的键图后继封闭做半径阶梯实验（$r=2..6$）：$|K^*|=99\to294\to939\to2863\to8879$，每加一半径约 $\times3.1$，危险键命中同步增长（$2\to96$）；复跑两次逐位复现。故固定半径的局部抽象不存在有限不动点〔kclosure〕。
\end{theorem}

研究域定位：残余为 lap/era 仿射系统的轨迹避让，与 cm2/M1 残余同型（类比归类；逐台形式重述未做）。
