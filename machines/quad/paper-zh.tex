\section{停机判据（每台独立，转移级完整证明）}

\begin{theorem}[cm16]\label{thm:h16}
cm16 停机 $\iff$ 某时刻状态 $D$ 读格 $w$ 之 $0$ 且 $\tau(w{+}1)=1$。
\end{theorem}
\begin{proof}
停机转移为 $F$ 读 $1$（$F1=\HALT$）。$F$ 的唯一入边为 $D0=\texttt{1RF}$：$D$ 于格 $w$ 读 $0$、写 $1$ 于 $w$、右行至 $w{+}1$ 入 $F$；故 $F$ 的每次到达都紧随一次 $D0$，且 $F$ 所读格恰为 $w{+}1$。于是停机发生 $\iff$ 某次 $D0$ 之后 $\tau(w{+}1)=1$，即所述判据。（$D0$ 的触发条件 $\tau(w)=0$ 已含在``$D$ 读格 $w$ 之 $0$''中。）两方向皆由这一条入边链直接给出。
\end{proof}

\begin{theorem}[cm26]\label{thm:h26}
cm26 停机 $\iff$ 某时刻状态 $B$ 读格 $q$ 之 $1$ 且 $\tau(q{-}1)=0$。
\end{theorem}
\begin{proof}
停机转移为 $E$ 读 $0$。$E$ 的唯一入边为 $B1=\texttt{0LE}$：$B$ 于 $q$ 读 $1$、写 $0$ 于 $q$、\textbf{左}行至 $q{-}1$ 入 $E$；$E$ 所读格为 $q{-}1$。停机 $\iff$ 某次 $B1$ 之后 $\tau(q{-}1)=0$。
\end{proof}

\begin{theorem}[cm51]\label{thm:h51}
cm51 停机 $\iff$ 某时刻状态 $C$ 或 $D$ 读格 $w$ 之 $0$ 且 $\tau(w{+}1)=1$。
\end{theorem}
\begin{proof}
停机转移为 $E$ 读 $1$。$E$ 的入边恰两条：$C0=\texttt{1RE}$ 与 $D0=\texttt{0RE}$，二者皆\textbf{右}行（写后至 $w{+}1$ 入 $E$）。故停机 $\iff$ 某次 $C0$ 或 $D0$ 之后 $\tau(w{+}1)=1$。
\end{proof}

\begin{theorem}[cm52]\label{thm:h52}
cm52 停机 $\iff$ 某时刻状态 $C$ 读格 $c$ 之 $0$ 且 $\tau(c{-}1)=0$。
\end{theorem}
\begin{proof}
停机转移为 $E$ 读 $0$。$E$ 的唯一入边为 $C0=\texttt{0LE}$：$C$ 于 $c$ 读 $0$、左行至 $c{-}1$ 入 $E$，$E$ 读 $\tau(c{-}1)$。
\end{proof}

\begin{remark}
四条判据的证明是纯转移级的（单入边或双入边枚举），不依赖任何动力学；校验器的作用是把判据表述与全部构型实例双向核对（防表述笔误），非证明的必要部分。
\end{remark}

\section{lap 文法与时代代数}

\begin{definition}[标记态与时代]
标记态 = 舞族例程结束时的 $(A,h,\tau)$；相邻标记态之间为一个时代。空带起始段经直接模拟进入首次标记。
\end{definition}

\begin{lemma}[标准 lap 字]\label{lem:lap}
对每台机器、每个参数 $R\ge1$，一个 lap 宏步的转移字为
\[
A0\,F0 \;\Big|\; B1^{R-2}\,B0\,C0^2\,C1\,D1\,A1 \;\Big|\; E1^{R-1}\,E0\,D1\,A0,
\]
共 $2R+14$ 步；净效应：左库右缘 $-1$（$A0$ 段擦）、右库 $-2$（$B$ 段扫掠）、隔三格存入 $+1$（$E$ 段回写），头净左移一格，即簿记 $(x,R,z)\mapsto(x{+}1,\,R{-}2,\,z{+}1)$。
\end{lemma}
\begin{proof}
字的每一段逐符号读自转移表：首段 $A0F0$ 进入 $F$-扫描；中段 $B1^{R-2}B0$ 为 $B$ 对右库的逐格扫描（读 $1$ 保写右行、遇 $0$ 分支），接四态回程 $C0^2C1D1A1$；末段 $E1^{R-1}E0D1A0$ 为 $E$ 的回写扫描与冲刺回 $A$。总步数 $2R+14$ 与净效应 $(x,R,z)\mapsto(x{+}1,R{-}2,z{+}1)$ 为校验器结论：对 $R\in\{1,\dots,40\}$ 的全部剩余类与边界、四台机器各自的深扫锚点（合计约 78 万 lap）逐格 cell-diff 验证〔casecheck-island，39 断言全过〕。
\end{proof}

\begin{remark}[时代结构（已验证事实）]
逐 lap 簿记由引理~\ref{lem:lap} 给出（每 lap 右库 $-2$）。库存耗尽触发排空与 churn 重置；重置分支的键集（键 =（位置类， 邻域块数））经深扫采集：cm16/26/51 各 78 键逐键相同、cm52 为 89 键。本文不声明时代层的闭式映射——逐键的输出表以 census/era 数据为准（复跑入口见文末）。
\end{remark}

\section{四台间已验证的对应事实}

\begin{remark}
以下为分层事实，非同构定理（四台转移表各异，转移级同构不存在）：
\begin{enumerate}
\item lap 文法与簿记在四台上同一（引理~\ref{lem:lap}，每台各自验证）；
\item 大时代 $(L_0,R_0)$ 序列 562 对在常量偏移（$0$ 或 $+1$）下零错位——深扫前缀上的观测〔$5\times10^9$ 步级〕；
\item 末端构型仅冻结骨架逐位相同，工作三元组差 $\pm100$（对早期``逐位相同''测量结论的修正）；
\item 停机判据不同源（定理~\ref{thm:h16}--\ref{thm:h52}：四台的停机态、入边与图案方向互不相同），非停机装配逐台独立进行。
\end{enumerate}
\end{remark}

\section{孤岛不变量与危险图案}

\begin{definition}[孤岛]
孤岛 = 图案 $(0^1,1^1,0^1)$（单格 $1$ 岛）。危险事件 = 该图案出现且\textbf{头远离}（与摧毁波不同位）。
\end{definition}

\begin{theorem}[孤岛不变量]\label{thm:island}
在验证域内（每台 $2\times10^7$ 步，892 个孤岛时刻）：孤岛每次出现均与摧毁波同位（头距 $\le1$）；孤岛的出现为瞬态——每个 lap 内孤岛存在于\textbf{两}个段（修正早期``仅边界 $R{=}1$ 段''的说法），且创造者与摧毁者为同一道经过波，存在窗口短于一个 lap〔casecheck-island，39 断言全过〕。
\end{theorem}

\begin{theorem}[词表内安全]\label{thm:vocab}
在已验证例程词表覆盖的一切 $A$-态落点处，定理~\ref{thm:h16}--\ref{thm:h52} 的图案前提不出现：冲刺落点的右间隙 $\ge2$（落点几何由字结构给出）；孤岛位被定理~\ref{thm:island} 排除；排空分支的落点窗口经 census 逐格核对。
\end{theorem}

\section{归约与开放问题}

\begin{theorem}[归约（每台）]\label{thm:reduction}
对每台机器 $m\in\{16,26,51,52\}$，以下等价：
\begin{enumerate}
\item $m$ 停机；
\item $m$ 的判据图案（定理~\ref{thm:h16}--\ref{thm:h52} 之二格条件）在某个\textbf{词表外}落点处成立（危险事件）。
\end{enumerate}
机器验证：audit 模式对词表外落点逐步五格检查（$10^9$ 步 $\times$4）危险出现 $=0$；guard 模式（$2\times10^{11}$ 步 $\times$4，逐步检查停机前提）零触发；裸机深扫 cm16 至 $5\times10^{11}$ 步零停机。
\end{theorem}
\begin{proof}
(2)$\Rightarrow$(1)：判据图案成立即触发定理~\ref{thm:h16}--\ref{thm:h52} 的入边链，停机。(1)$\Rightarrow$(2)：设停机，则某判据图案被访问；由定理~\ref{thm:vocab}（词表内安全），该访问必在词表覆盖之外的 $A$-态落点处；词表外落点的枚举完备性由 census/era 键集覆盖承载〔A$^-$〕。
\end{proof}

\begin{conjecture}[五格避让]\label{conj:quad}
四台的初始标记时代轨道 $(\mathfrak M^{(m)}_k)$ 满足 $\forall k$：时代 $\mathfrak M^{(m)}_k$ 内部无头远孤岛访问。
\end{conjecture}

\begin{theorem}[局部抽象否证]\label{thm:kclosure}
对 cm16 的键图后继封闭做半径阶梯实验（$r=2..6$）：$|K^*|=99\to294\to939\to2863\to8879$，每加一半径约 $\times3.1$，危险键命中同步增长（$2\to96$）；复跑两次逐位复现。故固定半径的局部抽象不存在有限不动点〔kclosure〕。
\end{theorem}

研究域定位：残余为 lap/era 仿射系统的轨迹避让，与 cm2/M1 残余同型（类比归类；逐台形式重述未做）。
