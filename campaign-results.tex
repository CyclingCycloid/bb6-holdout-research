\section{M1：结构理论与障碍定理}

\subsection{结构}

\subsection{无条件结果}
\begin{theorem}[停机不可达]
停机构型 $G(a{=}3)$ 结构性不可达：轨道上一切 $G$-头满足 $a+1=2^j\cdot(\text{奇}\ge3)$；支配性对 $\pm1$ 扰动稳健（2980/2980 反事实闭合；对抗剖面 2-adic 不可实现且从不停机）。
\end{theorem}

\begin{theorem}[包络]
类 $\le 2000$ 穷尽（基础表 1024 + 扩展 976/976）$\Rightarrow$ 按记录时刻表经验律（$\times4$/级），$\gtrsim 2^{2^{2000}}$ 步不停机。
\end{theorem}

\subsection{障碍（墙证硬）}
\begin{theorem}[周期排除]
签名动力学一阶有限状态但非最终周期：Brent $1.25\times10^9$ 步无环；$2.4\times10^6$ 梯残差两两互异；记录时刻表每级 $\times4$ 爬升。任何 $K$-位闭合将压制时刻表于深度 $K$ 以下——与爬升矛盾。
\end{theorem}
假设 M $\equiv$ 记录时刻表延续 $\equiv$ $\nu_2$-混沌（不可有限化）。
