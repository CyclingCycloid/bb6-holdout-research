\section{M1：结构理论与障碍}

\subsection{已证结构}
\begin{itemize}
\item 相位演算：九规则、精确步代价；与裸机 oracle 916 锚点逐步全等（含 13 个停机锚点双向一致）\grA{}；转移表经独立手工转写对拍（0--200 步零分歧、$10^{5,6,7}$ 检查点全同、停机锚 13025 步复现）\grA{}。
\item 序曲定理：$P(n)\to R(1,2n{+}2)\to G(n{+}2,n{+}1)\to^{m}\Phi(w,m)$，减半链闭式 $a_i+1=2^{m-i}w$——完整书面代数 \grA{}（另 1052 例机器复核）。
\item 轮律强归纳架构：级联闭式 \grA{}（头部代数）/尾部簿记 \grAm{}（$h\le60$ 逐游程验证）；阶梯类界 $A_j\le\tau_j^2$ 不等式链 \grA{}；主控段封闭：$h\le2000$ 穷尽（基础 1024 + 扩展 976/976）\grAm{}，抽样一致至 $1.2\times10^7$ \grB{}。
\item 支配引理：比较裕度 $W+Q-\theta$，$W\ge 3\cdot2^{2h-52-2\log_2h}\gg|Q|<2^{h+5}$，$h\ge82$ 恒正（代数 + 12,000 条仪表记录佐证常数）\grAm{}；$h\le81$ 逐点 \grB{}。
\item $\Delta t\le40$：终点子结构论证 \grAm{}（观测域 $[2,9]$）。
\end{itemize}

\subsection{无条件结果}
\begin{theorem}[停机不可达]\grA{}
轨道上一切 $G$-态头部满足 $a+1=2^{j}\cdot(\text{奇}\ge3)$，故 $G(a{=}3)$ 不可达；代数族（序曲/级联/地标头）部分为纸面证明，主控路径形式集部分为 273,654 形式 2-adic 审计零危险。扰动稳健性：单/稀疏翻转反事实 2980/2980 安全闭合 \grB{}；对抗剖面经 2-adic 展开有限性论证排除且实测从不停机 \grAm{}。
\end{theorem}

\begin{theorem}[包络]\grA{}
一切轮类 $\le2000$ 的运行不停机（该域逐类穷尽）。步数换算：按记录时刻表 $\times4$/级爬升律（\grB{}：实测至第 32 级，$1.25\times10^9$ 残差核步），首次触及类 2001 需 $\sim2^{2000}$ 轮，即非停机 $\gtrsim 2^{2^{2000}}$ 步——\textbf{换算的计时部分为经验律，严密度低于类形式}。
\end{theorem}

\subsection{障碍}
\begin{lemma}[周期排除]\grAm{}
签名动力学一阶有限状态；验证范围内（$1.25\times10^9$ 步 Brent 无环 \grB{}；$2.4\times10^6$ 梯残差两两互异 \grB{}）非最终周期；且任何 $K$-位有限闭合将压制记录时刻表于深度 $K$ 以下，与（观测未终止的）爬升矛盾（结构论证）。故假设 M 的内容 = $\nu_2$-混沌的延续，本架构内不可有限闭合。
\end{lemma}
另：不变量合成路线的不可能性（必要精度定理）\grAm{}——固定模数/线性模板在 $S(2,2,2)\to G(3,3)$ 四边反例链（逐边引擎复核）下不封闭；对抗剖面 2-adic 不可实现性论证 \grAm{}。
